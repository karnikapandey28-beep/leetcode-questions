=prev;
                prev=curr;
                curr=next;
            }
            //prev=new head of reversed group
            if(prevNode ==null)
            {
                head=prev;
            }
            else
            {
                prevNode.next=prev;
            }
            //temp is now the lastNode of the reversed group
            temp.next=nextNode;
            prevNode=temp;

            //Move to next group
            temp=nextNode;
        }
        return head;   
    }
