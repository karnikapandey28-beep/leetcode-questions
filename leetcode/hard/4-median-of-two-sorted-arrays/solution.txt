   //Copy remaining elements from nums1
        while(i<m)
        {
            nums3[k]=nums1[i];
            i++;
            k++;
        }
        //Copy remaining elements from nums2
        while(j<n)
        {
            nums3[k]=nums2[j];
            j++;
            k++;
        }
        //Find median
        if(total%2==1)
        {
            return nums3[total/2];
        }
        else
        {
            int mid=total/2;
            return (nums3[mid-1] +  (double)nums3[mid])/2.0;
        }
