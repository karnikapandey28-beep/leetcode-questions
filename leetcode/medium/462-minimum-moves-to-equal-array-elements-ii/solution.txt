class Solution {
    public int minMoves2(int[] nums) {
        Arrays.sort(nums);
      int n=nums.length;
        int mid=n/2;
        int median=0;
        int diff=0;
        if((nums.length%2)!=0)
        {
            median=(nums[mid]);
        }
        else
        {
            median=(nums[mid-1]+nums[mid])/2;
        }
        for(int i=0;i<n;i++)
        {
            diff+=Math.abs(median-nums[i]);
        }
        return diff;    
    }
}
