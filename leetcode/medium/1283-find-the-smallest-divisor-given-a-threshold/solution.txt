
        int ans=0;
        for(int i=0;i<nums.length;i++)
        {
            high=Math.max(high,nums[i]);
        }
        while(low<=high)
        {
            int sum=0;     //so sum can reset
            int mid=low+(high-low)/2;
            for(int i=0;i<nums.length;i++)
            {
                sum+=(int) Math.ceil((double)nums[i]/mid);
            }
            if(sum<=threshold)
            {
                ans=mid;
                high=mid-1;
            }
            else if(sum>threshold)
            {
                low=mid+1;
            }
        }
        return ans;   
    }
