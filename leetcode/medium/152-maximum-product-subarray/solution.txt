axProduct(int[] nums) 
    {
        int max=nums[0];
        int min=nums[0];
        int ans=nums[0];

        for(int i=1;i<nums.length;i++)
        {
            int x=nums[i];

            int op1=x;
            int op2=x*max;
            int op3=x*min;

            int newMax = Math.max(op1,Math.max(op2,op3));
            int newMin=Math.min(op1,Math.min(op2,op3));

            max=newMax;
            min=newMin;

            ans=Math.max(ans,max);
        }
        return ans;    
    }
}
