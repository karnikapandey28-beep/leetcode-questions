class Solution {
    public ListNode rotateRight(ListNode head, int k) 
    {
        if(head==null || head.next==null)
        {
            return head;
        }

        //Calculate length

        int len=1;
        ListNode tail=head;
        while(tail.next!=null)
        {
            tail=tail.next;
            len++;
        }

        //Handle large k

        k=k%len;
        if(k==0)
        {
            return head;
        }
