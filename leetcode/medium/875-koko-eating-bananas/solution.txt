class Solution {
    public int minEatingSpeed(int[] piles, int h) 
    {
        int low=1;
        int high=0;
        int ans=0;
        for(int i=0;i<piles.length;i++)
        {
            high=Math.max(high,piles[i]);
        }
        while(low<=high)
        {
            int mid=low+(high-low)/2;
            long sum=0;
            for(int i=0;i<piles.length;i++)
            {
                sum+=(int) Math.ceil((double)piles[i]/mid);
            }
            if(sum<=h)
            {
                ans=mid;
                high=mid-1;
            }
            else
            {
                low=mid+1;
            }
        }
        return ans;
    }
}
