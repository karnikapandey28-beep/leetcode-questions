leteMiddle(ListNode head) 
    {
        //Only one node

        if(head.next==null)
        {
            return null;
        }
        //Find length
        int n=0;
        ListNode temp=head;
        while(temp!=null)
        {
            n++;
            temp=temp.next;
        }
        //Reach node before middle
        temp=head;
        for(int i=1;i<n/2;i++)
        {
            temp=temp.next;
        }
        //Delete middle
        temp.next=temp.next.next;
        return head;    
    }
}
