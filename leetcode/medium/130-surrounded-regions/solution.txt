s(board,n,m,i,0);
            }
        }
        //Last column
        for(int i=0;i<n;i++)
        {
            if(board[i][m-1]=='O')
            {
                dfs(board,n,m,i,m-1);
            }
        }
        //Convert surrounded 0 to X and restore # to 0 
        for(int i=0;i<n;i++)
        {
            for(int j=0;j<m;j++)
            {
                if(board[i][j]=='#')
                {
                    board[i][j]='O';
                }
                else if(board[i][j] == 'O')
                {
                    board[i][j]='X';
                }
            }
        }
    }
