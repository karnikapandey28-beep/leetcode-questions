       int n=isConnected.length;
        boolean[] vis=new boolean[n];
        int count=0;

        for(int i=0;i<n;i++)
        {
        if(!vis[i])
        {
            count+=1;

            Queue<Integer>q=new LinkedList<>();

            q.add(i);
            vis[i]=true;

            while(!q.isEmpty())
            {
                int x=q.poll();
                for(int j=0;j<n;j++)
                {
                    if(x!=j && isConnected[x][j]==1 && !vis[j])
                    {
                        vis[j]=true;
                        q.add(j);
                    }
                }
            }
        }
        }
        return count;                 
    }
