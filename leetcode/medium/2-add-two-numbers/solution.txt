 */
class Solution {
    public ListNode addTwoNumbers(ListNode l1, ListNode l2) 
    {
        int len1=0;
        int len2=0;
        //Calculate length 
        for(ListNode p=l1;p!=null;p=p.next)
        {
            len1+=1;
        }
        for(ListNode p=l2;p!=null;p=p.next)
        {
            len2+=1;
        }
        int maxlen=Math.max(len1,len2);

        ListNode dummy=new ListNode(0);
        ListNode curr=dummy;

        int carry=0;
        for(int i=0;i<maxlen || carry!=0;i++)
        {
            int x=0;
            int y=0;
            if(l1!=null)
            {
                x=l1.val;
                l1=l1.next;
            }
            if(l2!=null)
            {
                y=l2.val;
                l2=l2.next;
            }
            int sum =x+y+carry;

            carry=sum/10;
            int digit=sum%10;

            curr.next=new ListNode(digit);
            curr=curr.next;
        }
        return dummy.next;    
    }
}
