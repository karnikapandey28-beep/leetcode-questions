class Solution {
    public int mySqrt(int x) 
    {
        int low=0;
        int high=x/2+1;
        int ans=0;
        while(low<=high)
        {
            int mid=low+(high-low)/2;
            long square=(long) mid*mid;
            if(square==x)
            {
                return mid;
            }
            else if(square<x)
            {
                low=mid+1;
                ans=mid;
            }
            else if(square>x)
            {
                high=mid-1;
            }
        }
        return ans;    
    }
}
