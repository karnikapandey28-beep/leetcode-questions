    val = x;
 *         next = null;
 *     }
 * }
 */
public class Solution {
    public ListNode getIntersectionNode(ListNode headA, ListNode headB) 
    {
        Map<ListNode,Integer>map=new HashMap<>();
        ListNode t=headA;
        while(t!=null)
        {
            map.put(t,1);
            t=t.next;
        }
        t=headB;
        while(t!=null)
        {
            if(map.containsKey(t))
            return t;
            t=t.next;
        }
        return null;        
    }
}
